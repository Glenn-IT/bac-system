Count Documents by Status:
SELECT status, COUNT(*) as count 
FROM eligibility_docs 
GROUP BY status;

View Expired Documents:
SELECT s.company_name, dt.document_name, ed.expiration_date
FROM eligibility_docs ed
JOIN suppliers s ON ed.supplier_id = s.id
JOIN doc_types dt ON ed.doc_type_id = dt.id
WHERE ed.status = 'Expired'
ORDER BY ed.expiration_date;

List All Activity Today:
SELECT * FROM activity_logs 
WHERE DATE(created_at) = CURDATE()
ORDER BY created_at DESC;

📊 COMPLIANCE CALCULATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Formula: (Valid Documents / Total Documents) × 100

Example:
Valid: 10
For Renewal: 2
Expired: 1
Missing: 0
Total: 13

Compliance = (10 / 13) × 100 = 76.9%

Rating:
≥ 80% → Excellent (Green bar)
50-79% → Fair (Yellow bar)
< 50% → Poor (Red bar)
